# Not on GitHub Actions? Any scheduler works. Keep the brief and the dead-man as SEPARATE jobs,
# on SEPARATE schedules, so a dead brief job cannot take its own alarm down with it.

# 13:30 every day: write the draft brief (arm seats one at a time)
30 13 * * *  cd /srv/exec-org && EXEC_ARMED=cfo,cto exec-org brief

# 16:10 every day: did a brief actually land? (also nudges about unarmed seats after EXEC_ARM_BY)
10 16 * * *  cd /srv/exec-org && EXEC_ARMED=cfo,cto EXEC_ALERT_WEBHOOK=https://hooks.example.com/your-channel exec-org deadman
